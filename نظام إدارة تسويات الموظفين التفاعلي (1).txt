import React, { useState, useEffect, useMemo, useRef } from 'react';
import { initializeApp } from 'firebase/app';
import { 
  getAuth, 
  signInAnonymously, 
  signInWithCustomToken, 
  onAuthStateChanged 
} from 'firebase/auth';
import { 
  getFirestore, 
  collection, 
  doc, 
  addDoc, 
  updateDoc, 
  deleteDoc, 
  onSnapshot 
} from 'firebase/firestore';
import { 
  UserPlus, 
  Search, 
  Trash2, 
  Edit3, 
  Download, 
  Upload, 
  CheckCircle2, 
  AlertTriangle, 
  FileSpreadsheet,

  IdCard,

} from 'lucide-react';

// ==========================================
// 1. إعدادات الاتصال السحابي والمزامنة الفورية
// ==========================================
const firebaseConfig = typeof __firebase_config !== 'undefined' ? JSON.parse(__firebase_config) : {};
const app = initializeApp(firebaseConfig);
const auth = getAuth(app);
const db = getFirestore(app);
const appId = typeof __app_id !== 'undefined' ? __app_id : 'employee-settlement-app';

// بيانات حقوق الملكية والبورتفوليو الثابتة
const OWNER_NAME = 'نورهان صلاح';
const PORTFOLIO_URL = 'https://nourhan-salah.github.io/nour-portfolio/';

export default function App() {
  // حالات المستخدم والاتصال
  const [user, setUser] = useState(null);
  const [isSyncing, setIsSyncing] = useState(true);
  const [syncError, setSyncError] = useState(null);

  // بيانات الموظفين من القاعدة اللحظية
  const [employees, setEmployees] = useState([]);

  // حالة حماية الكود المصدري (مفعلة دائماً)
  const [sourceProtectionEnabled] = useState(true);

  // حالات نموذج الإدخال والتعديل
  const [ledgerNo, setLedgerNo] = useState('');
  const [empName, setEmpName] = useState('');
  const [empId, setEmpId] = useState('');
  const [netSettlement, setNetSettlement] = useState('');
  const [notes, setNotes] = useState('');
  const [editingDocId, setEditingDocId] = useState(null);
  const [isSubmitting, setIsSubmitting] = useState(false);

  // حالات البحث والفلترة والترتيب
  const [searchQuery, setSearchQuery] = useState('');
  const [searchField, setSearchField] = useState('all'); // 'all', 'name', 'empId', 'ledgerNo'
  const [sortBy, setSortBy] = useState('createdAtDesc'); // 'createdAtDesc', 'ledgerAsc', 'nameAsc', 'amountDesc', 'amountAsc'

  // حالات التحديد المتعدد والحذف
  const [selectedIds, setSelectedIds] = useState([]);
  const [confirmDeleteItem, setConfirmDeleteItem] = useState(null); // { id, name, isBulk }

  // حالات الاستيراد من إكسل (لصق أو ملف)
  const [showImportModal, setShowImportModal] = useState(false);
  const [pasteData, setPasteData] = useState('');
  const [isImporting, setIsImporting] = useState(false);

  // حالات الإشعارات (Toast)
  const [toast, setToast] = useState(null);
  const fileInputRef = useRef(null);

  // دالة إظهار إشعار سريع
  const showNotification = (message, type = 'success') => {
    setToast({ message, type });
    setTimeout(() => {
      setToast((prev) => (prev?.message === message ? null : prev));
    }, 4000);
  };

  // ==========================================
  // 2. مصادقة المستخدم الفورية (Rule 3)
  // ==========================================
  useEffect(() => {
    const initAuth = async () => {
      try {
        if (typeof __initial_auth_token !== 'undefined' && __initial_auth_token) {
          await signInWithCustomToken(auth, __initial_auth_token);
        } else {
          await signInAnonymously(auth);
        }
      } catch (err) {
        console.error('Auth error:', err);
        setSyncError('تعذر الاتصال بخادم المصادقة');
      }
    };
    initAuth();
    const unsubscribe = onAuthStateChanged(auth, (currentUser) => {
      setUser(currentUser);
    });
    return () => unsubscribe();
  }, []);

  // ==========================================
  // 3. الاستماع اللحظي للبيانات المشتركة (Rule 1 & 2)
  // ==========================================
  useEffect(() => {
    if (!user) return;
    setIsSyncing(true);

    const colRef = collection(db, 'artifacts', appId, 'public', 'data', 'settlement_records');
    
    const unsubscribe = onSnapshot(
      colRef,
      (snapshot) => {
        const loaded = snapshot.docs.map((d) => ({
          id: d.id,
          ...d.data()
        }));
        setEmployees(loaded);
        setIsSyncing(false);
        setSyncError(null);
      },
      (error) => {
        console.error('Firestore sync error:', error);
        setSyncError('حدث خطأ أثناء المزامنة الحية للبيانات');
        setIsSyncing(false);
      }
    );

    return () => unsubscribe();
  }, [user]);

  // الاستماع اللحظي لإعدادات حقوق الملكية (Copyright) وحماية الكود المصدري ليظهر اسمك لدى الجميع
  useEffect(() => {
    if (!user) return;

    const settingsDocRef = doc(db, 'artifacts', appId, 'public', 'data', 'system_settings', 'copyright_config');
    const unsubscribe = onSnapshot(
      settingsDocRef,
      (docSnap) => {
        if (docSnap.exists()) {
          const data = docSnap.data();
          if (data.ownerName) setOwnerName(data.ownerName);
          if (data.ownerTitle !== undefined) setOwnerTitle(data.ownerTitle);
          if (data.ownerPinHash !== undefined) setOwnerPinHash(data.ownerPinHash);
          if (typeof data.sourceProtectionEnabled === 'boolean') {
            setSourceProtectionEnabled(data.sourceProtectionEnabled);
          }
        }
      },
      (err) => {
        console.error('Settings sync error:', err);
      }
    );

    return () => unsubscribe();
  }, [user]);

  // ==========================================
  // 3.5 نظام قفل وحماية الـ Source Code (Client-Side Protection)
  // ==========================================
  useEffect(() => {
    if (!sourceProtectionEnabled) return;

    const triggerSecurityAlert = (reason) => {
      showNotification(`🔒 عفواً، الكود المصدري وحقوق التصميم محمية (${reason}) — ${OWNER_NAME}`, 'error');
    };

    // منع كليك يمين (Context Menu)
    const handleContextMenu = (e) => {
      e.preventDefault();
      triggerSecurityAlert('قفل القائمة المنسدلة');
      return false;
    };

    // منع اختصارات لوحة المفاتيح الخاصة بفحص الكود أو حفظ الصفحة
    const handleKeyDown = (e) => {
      const key = e.key?.toUpperCase();
      const ctrlOrMeta = e.ctrlKey || e.metaKey;

      // F12
      if (e.key === 'F12' || e.keyCode === 123) {
        e.preventDefault();
        e.stopPropagation();
        triggerSecurityAlert('منع F12');
        return false;
      }

      // Ctrl+Shift+I / Ctrl+Shift+J / Ctrl+Shift+C (أدوات المطورين)
      if (ctrlOrMeta && e.shiftKey && ['I', 'J', 'C', 'K'].includes(key)) {
        e.preventDefault();
        e.stopPropagation();
        triggerSecurityAlert('قفل فحص العناصر');
        return false;
      }

      // Ctrl+U (عرض الكود المصدري View Source) أو Ctrl+S (حفظ الكود)
      if (ctrlOrMeta && ['U', 'S'].includes(key)) {
        e.preventDefault();
        e.stopPropagation();
        triggerSecurityAlert('حماية الكود المصدري');
        return false;
      }
    };

    // منع سحب العناصر
    const handleDragStart = (e) => {
      e.preventDefault();
    };

    document.addEventListener('contextmenu', handleContextMenu);
    document.addEventListener('keydown', handleKeyDown, true);
    document.addEventListener('dragstart', handleDragStart);

    return () => {
      document.removeEventListener('contextmenu', handleContextMenu);
      document.removeEventListener('keydown', handleKeyDown, true);
      document.removeEventListener('dragstart', handleDragStart);
    };
  }, [sourceProtectionEnabled]);

  // ==========================================
  // 4. التحقق الذكي من التكرار أثناء الكتابة
  // ==========================================
  const duplicateCheck = useMemo(() => {
    const cleanLedger = ledgerNo.trim();
    const cleanEmpId = empId.trim();
    const cleanName = empName.trim();

    let duplicateLedger = null;
    let duplicateEmpId = null;
    let duplicateName = null;

    for (const emp of employees) {
      if (editingDocId && emp.id === editingDocId) continue;

      if (cleanLedger && String(emp.ledgerNo).trim() === cleanLedger) {
        duplicateLedger = emp;
      }
      if (cleanEmpId && String(emp.empId).trim().toLowerCase() === cleanEmpId.toLowerCase()) {
        duplicateEmpId = emp;
      }
      if (cleanName && cleanName.length > 3 && String(emp.empName).trim() === cleanName) {
        duplicateName = emp;
      }
    }

    return { duplicateLedger, duplicateEmpId, duplicateName };
  }, [ledgerNo, empId, empName, employees, editingDocId]);

  // ==========================================
  // 5. دوال الإضافة والتعديل والحذف
  // ==========================================
  const resetForm = () => {
    setLedgerNo('');
    setEmpName('');
    setEmpId('');
    setNetSettlement('');
    setNotes('');
    setEditingDocId(null);
  };

  const handleFormSubmit = async (e) => {
    e.preventDefault();
    if (!user) return;

    const cleanLedger = ledgerNo.trim();
    const cleanName = empName.trim();
    const cleanId = empId.trim();
    const parsedAmount = parseFloat(netSettlement);

    if (!cleanLedger || !cleanName || !cleanId || isNaN(parsedAmount)) {
      showNotification('يرجى ملء جميع الحقول الأساسية بشكل صحيح', 'error');
      return;
    }

    setIsSubmitting(true);
    try {
      const colRef = collection(db, 'artifacts', appId, 'public', 'data', 'settlement_records');
      
      if (editingDocId) {
        const docRef = doc(db, 'artifacts', appId, 'public', 'data', 'settlement_records', editingDocId);
        await updateDoc(docRef, {
          ledgerNo: cleanLedger,
          empName: cleanName,
          empId: cleanId,
          netSettlement: parsedAmount,
          notes: notes.trim(),
          updatedAt: Date.now(),
          updatedBy: user.uid
        });
        showNotification(`تم تحديث بيانات "${cleanName}" بنجاح`);
        resetForm();
      } else {
        await addDoc(colRef, {
          ledgerNo: cleanLedger,
          empName: cleanName,
          empId: cleanId,
          netSettlement: parsedAmount,
          notes: notes.trim(),
          createdAt: Date.now(),
          updatedAt: Date.now(),
          createdBy: user.uid
        });
        showNotification(`تمت إضافة الموظف "${cleanName}" إلى الشيت التفاعلي`);
        resetForm();
      }
    } catch (error) {
      console.error('Submit error:', error);
      showNotification('حدث خطأ أثناء حفظ البيانات، يرجى المحاولة مرة أخرى', 'error');
    } finally {
      setIsSubmitting(false);
    }
  };

  const handleStartEdit = (emp) => {
    setEditingDocId(emp.id);
    setLedgerNo(String(emp.ledgerNo || ''));
    setEmpName(String(emp.empName || ''));
    setEmpId(String(emp.empId || ''));
    setNetSettlement(String(emp.netSettlement ?? ''));
    setNotes(String(emp.notes || ''));
    window.scrollTo({ top: 0, behavior: 'smooth' });
  };

  const executeDelete = async () => {
    if (!user || !confirmDeleteItem) return;

    try {
      if (confirmDeleteItem.isBulk) {
        const idsToDelete = [...selectedIds];
        await Promise.all(
          idsToDelete.map((id) =>
            deleteDoc(doc(db, 'artifacts', appId, 'public', 'data', 'settlement_records', id))
          )
        );
        setSelectedIds([]);
        showNotification(`تم مسح ${idsToDelete.length} سجل من الشيت بنجاح`);
      } else {
        await deleteDoc(
          doc(db, 'artifacts', appId, 'public', 'data', 'settlement_records', confirmDeleteItem.id)
        );
        setSelectedIds((prev) => prev.filter((item) => item !== confirmDeleteItem.id));
        if (editingDocId === confirmDeleteItem.id) resetForm();
        showNotification(`تم مسح الموظف "${confirmDeleteItem.name}" من الشيت نهائياً`);
      }
    } catch (error) {
      console.error('Delete error:', error);
      showNotification('حدث خطأ أثناء الحذف', 'error');
    } finally {
      setConfirmDeleteItem(null);
    }
  };

  // إضافة بيانات تجريبية سريعة للمعاينة
  const handleAddSampleData = async () => {
    if (!user) return;
    const samples = [
      { ledgerNo: '101', empName: 'أحمد محمود عبد العزيز', empId: 'EMP-4091', netSettlement: 14500.50, notes: 'تسوية نهاية الخدمة كاملة' },
      { ledgerNo: '102', empName: 'سارة إبراهيم الكيلاني', empId: 'EMP-4092', netSettlement: 8250.00, notes: 'شامل رصيد الإجازات' },
      { ledgerNo: '105', empName: 'محمد طارق الدمرداش', empId: 'EMP-4105', netSettlement: 21300.75, notes: 'مكافأة + تسوية مالية' },
      { ledgerNo: '108', empName: 'نهى عبد الرحمن فؤاد', empId: 'EMP-4118', netSettlement: 6400.00, notes: 'تم المراجعة' },
    ];

    try {
      const colRef = collection(db, 'artifacts', appId, 'public', 'data', 'settlement_records');
      await Promise.all(
        samples.map((s) =>
          addDoc(colRef, {
            ...s,
            createdAt: Date.now(),
            updatedAt: Date.now(),
            createdBy: user.uid
          })
        )
      );
      showNotification('تمت إضافة 4 سجلات نموذجية للتجربة');
    } catch (e) {
      showNotification('تعذر إضافة البيانات التجريبية', 'error');
    }
  };

  // ==========================================
  // 6. الفلترة والبحث الفوري والترتيب
  // ==========================================
  const filteredAndSortedEmployees = useMemo(() => {
    const q = searchQuery.trim().toLowerCase();

    const filtered = employees.filter((emp) => {
      if (!q) return true;
      const nameStr = String(emp.empName || '').toLowerCase();
      const idStr = String(emp.empId || '').toLowerCase();
      const ledgerStr = String(emp.ledgerNo || '').toLowerCase();
      const notesStr = String(emp.notes || '').toLowerCase();

      if (searchField === 'name') return nameStr.includes(q);
      if (searchField === 'empId') return idStr.includes(q);
      if (searchField === 'ledgerNo') return ledgerStr.includes(q);

      return (
        nameStr.includes(q) ||
        idStr.includes(q) ||
        ledgerStr.includes(q) ||
        notesStr.includes(q)
      );
    });

    return filtered.sort((a, b) => {
      if (sortBy === 'ledgerAsc') {
        return String(a.ledgerNo).localeCompare(String(b.ledgerNo), undefined, { numeric: true });
      }
      if (sortBy === 'nameAsc') {
        return String(a.empName).localeCompare(String(b.empName), 'ar');
      }
      if (sortBy === 'amountDesc') {
        return (Number(b.netSettlement) || 0) - (Number(a.netSettlement) || 0);
      }
      if (sortBy === 'amountAsc') {
        return (Number(a.netSettlement) || 0) - (Number(b.netSettlement) || 0);
      }
      // Default: createdAtDesc
      return (b.createdAt || 0) - (a.createdAt || 0);
    });
  }, [employees, searchQuery, searchField, sortBy]);

  // إحصائيات حية
  const stats = useMemo(() => {
    const totalCount = employees.length;
    const totalSettlement = employees.reduce((acc, cur) => acc + (Number(cur.netSettlement) || 0), 0);
    const filteredCount = filteredAndSortedEmployees.length;
    const filteredSettlement = filteredAndSortedEmployees.reduce(
      (acc, cur) => acc + (Number(cur.netSettlement) || 0),
      0
    );

    return { totalCount, totalSettlement, filteredCount, filteredSettlement };
  }, [employees, filteredAndSortedEmployees]);

  // ==========================================
  // 7. التصدير والاستيراد من وإلى Excel (CSV UTF-8 BOM)
  // ==========================================
  const handleExportExcelCSV = () => {
    if (filteredAndSortedEmployees.length === 0) {
      showNotification('لا توجد بيانات لتصديرها حالياً', 'error');
      return;
    }

    // إضافة BOM لضمان فتح الملف باللغة العربية السليمة في برنامج Microsoft Excel
    const BOM = '\uFEFF';
    const headers = ['رقم الموظف بالدفتر', 'اسم الموظف', 'ID الموظف', 'صافي التسوية', 'ملاحظات'];
    
    const rows = filteredAndSortedEmployees.map((emp) => [
      `"${String(emp.ledgerNo || '').replace(/"/g, '""')}"`,
      `"${String(emp.empName || '').replace(/"/g, '""')}"`,
      `"${String(emp.empId || '').replace(/"/g, '""')}"`,
      Number(emp.netSettlement || 0).toFixed(2),
      `"${String(emp.notes || '').replace(/"/g, '""')}"`
    ]);

    // تضمين توقيع حقوق الملكية باسم نورهان صلاح في نهاية شيت الإكسل المصدّر
    const copyrightRow = [
      `"جميع الحقوق محفوظة © ${new Date().getFullYear()}"`,
      `"تصميم وتطوير: ${OWNER_NAME}"`,
      `"${PORTFOLIO_URL}"`,
      `""`,
      `"نظام محمي"`
    ];

    const csvContent =
      BOM +
      [headers.join(','), ...rows.map((r) => r.join(',')), '', copyrightRow.join(',')].join('\r\n');
    const blob = new Blob([csvContent], { type: 'text/csv;charset=utf-8;' });
    const url = URL.createObjectURL(blob);

    const link = document.createElement('a');
    link.href = url;
    const dateStr = new Date().toISOString().slice(0, 10);
    link.setAttribute('download', `شيت_تسويات_الموظفين_${dateStr}.csv`);
    document.body.appendChild(link);
    link.click();
    document.body.removeChild(link);
    URL.revokeObjectURL(url);

    showNotification(`تم تحميل شيت الإكسل موثقاً بحقوق الملكية لـ "${OWNER_NAME}"`);
  };

  // تحليل النص الملصق من إكسل أو المرفوع من ملف CSV
  const processImportedText = async (rawText) => {
    if (!user || !rawText.trim()) return;
    setIsImporting(true);

    try {
      const lines = rawText
        .split(/\r?\n/)
        .map((l) => l.trim())
        .filter(Boolean);

      const recordsToInsert = [];

      for (let i = 0; i < lines.length; i++) {
        const line = lines[i];
        // الفصل بواسطة Tab (نسخ مباشر من إكسل) أو فاصلة , أو فاصلة منقوطة ;
        const parts = line.includes('\t')
          ? line.split('\t')
          : line.includes(';')
          ? line.split(';')
          : line.split(',');

        const cleaned = parts.map((p) => p.replace(/^["']|["']$/g, '').trim());
        if (cleaned.length < 3) continue;

        // تخطي سطر العناوين إن وجد
        if (
          i === 0 &&
          (cleaned[0].includes('رقم') ||
            cleaned[1].includes('اسم') ||
            cleaned[2].toLowerCase().includes('id'))
        ) {
          continue;
        }

        const lNo = cleaned[0] || '-';
        const eName = cleaned[1] || 'بدون اسم';
        const eId = cleaned[2] || '-';
        const rawAmount = cleaned[3] ? cleaned[3].replace(/[^0-9.-]/g, '') : '0';
        const amount = parseFloat(rawAmount) || 0;
        const note = cleaned[4] || '';

        recordsToInsert.push({
          ledgerNo: lNo,
          empName: eName,
          empId: eId,
          netSettlement: amount,
          notes: note
        });
      }

      if (recordsToInsert.length === 0) {
        showNotification('لم يتم العثور على صفوف صالحة للاستيراد، تأكد من ترتيب الأعمدة', 'error');
        setIsImporting(false);
        return;
      }

      const colRef = collection(db, 'artifacts', appId, 'public', 'data', 'settlement_records');
      await Promise.all(
        recordsToInsert.map((rec) =>
          addDoc(colRef, {
            ...rec,
            createdAt: Date.now(),
            updatedAt: Date.now(),
            createdBy: user.uid
          })
        )
      );

      showNotification(`تم استيراد ${recordsToInsert.length} موظف إلى الشيت التفاعلي بنجاح!`);
      setPasteData('');
      setShowImportModal(false);
    } catch (err) {
      console.error('Import error:', err);
      showNotification('حدث خطأ أثناء استيراد البيانات', 'error');
    } finally {
      setIsImporting(false);
    }
  };

  const handleFileUpload = (e) => {
    const file = e.target.files?.[0];
    if (!file) return;

    const reader = new FileReader();
    reader.onload = (event) => {
      const content = event.target?.result;
      if (typeof content === 'string') {
        setPasteData(content);
      }
    };
    reader.readAsText(file, 'UTF-8');
  };

  // تنسيق الأرقام المالية
  const formatCurrency = (val) => {
    const num = Number(val);
    if (isNaN(num)) return '0.00';
    return num.toLocaleString('ar-EG', {
      minimumFractionDigits: 2,
      maximumFractionDigits: 2
    });
  };

  // التحديد الجماعي
  const toggleSelectAll = () => {
    if (selectedIds.length === filteredAndSortedEmployees.length) {
      setSelectedIds([]);
    } else {
      setSelectedIds(filteredAndSortedEmployees.map((e) => e.id));
    }
  };

  const toggleSelectRow = (id) => {
    setSelectedIds((prev) =>
      prev.includes(id) ? prev.filter((item) => item !== id) : [...prev, id]
    );
  };

  // هل النموذج يحتوي على بيانات حالياً للمعاينة الحية؟
  const hasLiveInput = Boolean(
    ledgerNo.trim() || empName.trim() || empId.trim() || netSettlement.trim() || notes.trim()
  );

  return (
    <div
      dir="rtl"
      onContextMenu={(e) => {
        if (sourceProtectionEnabled) e.preventDefault();
      }}
      className={`min-h-screen bg-slate-950 text-slate-100 font-sans pb-16 selection:bg-emerald-500 selection:text-white flex flex-col justify-between ${
        sourceProtectionEnabled ? 'select-none' : ''
      }`}
    >
      {/* استيراد خط Cairo العربي ليبدو التصميم عصرياً واحترافياً */}
      <style>{`
        @import url('https://fonts.googleapis.com/css2?family=Cairo:wght@400;500;600;700;800&display=swap');
        body, * {
          font-family: 'Cairo', sans-serif;
        }
        input, textarea, select {
          user-select: text !important;
          -webkit-user-select: text !important;
        }
        @keyframes pulse-border {
          0%, 100% { border-color: rgba(16, 185, 129, 0.8); }
          50% { border-color: rgba(16, 185, 129, 0.25); }
        }
        .live-row-glow {
          animation: pulse-border 2s infinite;
        }
      `}</style>

      {/* إشعار عائم (Toast Notification) */}
      {toast && (

            {/* جدول شيت الإكسل التفاعلي */}
            <div className="bg-slate-900 border border-slate-800 rounded-3xl shadow-xl overflow-hidden">
              <div className="px-5 py-4 border-b border-slate-800 flex flex-wrap items-center justify-between gap-2 bg-slate-900/60">
                <div className="flex items-center gap-2">
                  <FileSpreadsheet className="w-5 h-5 text-emerald-400" />
                  <h3 className="font-bold text-white text-sm sm:text-base">
                    شيت الإكسل التفاعلي المشترك
                  </h3>
                  <span className="text-xs bg-slate-800 text-slate-300 px-2.5 py-0.5 rounded-full font-semibold">
                    {filteredAndSortedEmployees.length} سجل
                  </span>
                </div>

                {employees.length === 0 && (
                  <button
                    onClick={handleAddSampleData}
                    className="text-xs font-bold text-emerald-400 hover:text-emerald-300 bg-emerald-500/10 border border-emerald-500/20 px-3 py-1.5 rounded-xl flex items-center gap-1.5 cursor-pointer"
                  >
                    <Sparkles className="w-3.5 h-3.5" />
                    <span>إضافة بيانات تجريبية للمعاينة</span>
                  </button>
                )}
              </div>

              <div className="overflow-x-auto">
                <table className="w-full text-right border-collapse">
                  <thead>
                    <tr className="bg-slate-950/80 text-slate-400 text-xs uppercase border-b border-slate-800">
                      <th className="py-3.5 px-4 w-10 text-center">
                        <input
                          type="checkbox"
                          checked={
                            filteredAndSortedEmployees.length > 0 &&
                            selectedIds.length === filteredAndSortedEmployees.length
                          }
                          onChange={toggleSelectAll}
                          className="rounded border-slate-700 bg-slate-900 text-emerald-500 focus:ring-emerald-500 cursor-pointer"
                        />
                      </th>
                      <th className="py-3.5 px-3 font-bold">رقم الدفتر</th>
                      <th className="py-3.5 px-4 font-bold">اسم الموظف</th>
                      <th className="py-3.5 px-4 font-bold">الـ ID التعريفي</th>
                      <th className="py-3.5 px-4 font-bold">صافي التسوية</th>
                      <th className="py-3.5 px-4 font-bold">ملاحظات</th>
                      <th className="py-3.5 px-4 font-bold text-center">إجراءات (تعديل / مسح)</th>
                    </tr>
                  </thead>

                  <tbody className="divide-y divide-slate-800/70 text-sm">
                    {/* صف المعاينة الحية داخل الجدول أثناء الكتابة! */}
                    {hasLiveInput && !editingDocId && (
                      <tr className="bg-emerald-950/30 border-2 border-dashed border-emerald-500/60 live-row-glow">
                        <td className="py-3 px-4 text-center">
                          <span className="inline-block w-2.5 h-2.5 rounded-full bg-emerald-400 animate-ping" />
                        </td>
                        <td className="py-3 px-3 font-bold text-amber-300">
                          {ledgerNo || '...'}
                        </td>
                        <td className="py-3 px-4 font-bold text-white">
                          <div className="flex items-center gap-2">
                            <span>{empName || 'جاري كتابة الاسم...'}</span>
                            <span className="text-[10px] bg-emerald-500/20 text-emerald-300 px-2 py-0.5 rounded-md border border-emerald-500/30">
                              يكتب الآن...
                            </span>
                          </div>
                        </td>
                        <td className="py-3 px-4 font-mono text-xs text-indigo-300 font-bold">
                          {empId || '...'}
                        </td>
                        <td className="py-3 px-4 font-extrabold text-emerald-400">
                          {netSettlement ? `${formatCurrency(netSettlement)} ج.م` : '0.00 ج.م'}
                        </td>
                        <td className="py-3 px-4 text-xs text-slate-400">
                          {notes || '-'}
                        </td>
                        <td className="py-3 px-4 text-center text-xs text-emerald-400 font-semibold">
                          اضغط حفظ للإدراج
                        </td>
                      </tr>
                    )}

                    {/* عرض صفوف الموظفين الفعليين */}
                    {filteredAndSortedEmployees.length === 0 ? (
                      <tr>
                        <td colSpan={7} className="py-14 text-center text-slate-400">
                          <div className="max-w-sm mx-auto space-y-3">
                            <div className="w-14 h-14 rounded-2xl bg-slate-800/80 border border-slate-700 flex items-center justify-center mx-auto text-slate-400">
                              <Search className="w-7 h-7" />
                            </div>
                            <p className="font-bold text-base text-slate-200">
                              {searchQuery
                                ? `لا يوجد موظف مطابق لـ "${searchQuery}"`
                                : 'شيت التسويات فارغ حالياً'}
                            </p>
                            <p className="text-xs text-slate-400 leading-relaxed">
                              {searchQuery
                                ? 'تأكد من كتابة الاسم أو الـ ID أو رقم الدفتر بشكل صحيح، أو قم بمسح كلمة البحث لعرض كل الشيت.'
                                : 'ابدأ بإدخال بيانات الموظفين من النموذج الجانبي أو قم باستيراد شيت الإكسل الخاص بك مباشرة.'}
                            </p>
                          </div>
                        </td>
                      </tr>
                    ) : (
                      filteredAndSortedEmployees.map((emp) => {
                        const isSelected = selectedIds.includes(emp.id);
                        const isBeingEdited = editingDocId === emp.id;

                        return (
                          <tr
                            key={emp.id}
                            className={`transition-colors ${
                              isBeingEdited
                                ? 'bg-amber-500/15 border-r-4 border-amber-400'
                                : isSelected
                                ? 'bg-rose-500/10'
                                : 'hover:bg-slate-800/50'
                            }`}
                          >
                            <td className="py-3.5 px-4 text-center">
                              <input
                                type="checkbox"
                                checked={isSelected}
                                onChange={() => toggleSelectRow(emp.id)}
                                className="rounded border-slate-700 bg-slate-900 text-emerald-500 focus:ring-emerald-500 cursor-pointer"
                              />
                            </td>

                            {/* رقم الدفتر */}
                            <td className="py-3.5 px-3 whitespace-nowrap">
                              <span className="inline-flex items-center px-2.5 py-1 rounded-lg text-xs font-extrabold bg-amber-500/10 text-amber-300 border border-amber-500/20">
                                #{emp.ledgerNo}
                              </span>
                            </td>

                            {/* اسم الموظف */}
                            <td className="py-3.5 px-4 font-bold text-white">
                              {emp.empName}
                              {isBeingEdited && (
                                <span className="mr-2 text-[10px] bg-amber-500/20 text-amber-300 px-2 py-0.5 rounded-md">
                                  قيد التعديل...
                                </span>
                              )}
                            </td>

                            {/* ID الموظف */}
                            <td className="py-3.5 px-4 whitespace-nowrap">
                              <span className="font-mono text-xs font-bold px-2.5 py-1 rounded-lg bg-indigo-500/10 text-indigo-300 border border-indigo-500/20">
                                {emp.empId}
                              </span>
                            </td>

                            {/* صافي التسوية */}
                            <td className="py-3.5 px-4 whitespace-nowrap font-extrabold text-emerald-400">
                              {formatCurrency(emp.netSettlement)}{' '}
                              <span className="text-[11px] font-normal text-slate-400">ج.م</span>
                            </td>

                            {/* ملاحظات */}
                            <td className="py-3.5 px-4 text-xs text-slate-300 max-w-[160px] truncate">
                              {emp.notes || <span className="text-slate-600">-</span>}
                            </td>

                            {/* أزرار التعديل والمسح الفوري */}
                            <td className="py-3.5 px-4 whitespace-nowrap text-center">
                              <div className="flex items-center justify-center gap-2">
                                <button
                                  onClick={() => handleStartEdit(emp)}
                                  className="flex items-center gap-1 px-2.5 py-1.5 rounded-xl text-xs font-bold bg-slate-800 hover:bg-amber-500/20 text-slate-200 hover:text-amber-300 border border-slate-700 hover:border-amber-500/40 transition cursor-pointer"
                                  title="تعديل هذا السجل"
                                >
                                  <Edit3 className="w-3.5 h-3.5" />
                                  <span>تعديل</span>
                                </button>

                                <button
                                  onClick={() =>
                                    setConfirmDeleteItem({
                                      id: emp.id,
                                      name: emp.empName,
                                      ledger: emp.ledgerNo,
                                      isBulk: false
                                    })
                                  }
                                  className="flex items-center gap-1 px-2.5 py-1.5 rounded-xl text-xs font-bold bg-rose-500/10 hover:bg-rose-600 text-rose-400 hover:text-white border border-rose-500/20 transition cursor-pointer"
                                  title="مسح من شيت الإكسل"
                                >
                                  <Trash2 className="w-3.5 h-3.5" />
                                  <span>مسح</span>
                                </button>
                              </div>
                            </td>
                          </tr>
                        );
                      })
                    )}
                  </tbody>
                </table>
              </div>
            </div>
          </div>
        </div>
      </main>

      {/* ==========================================
          نافذة تأكيد المسح (Modal) بدلاً من alert/confirm
      ========================================== */}
      {confirmDeleteItem && (
        <div className="fixed inset-0 z-50 flex items-center justify-center bg-slate-950/80 backdrop-blur-sm p-4">
          <div className="bg-slate-900 border border-slate-800 rounded-3xl max-w-md w-full p-6 shadow-2xl space-y-5 animate-fadeIn">
            <div className="flex items-center gap-3.5">
              <div className="w-12 h-12 rounded-2xl bg-rose-500/15 border border-rose-500/30 flex items-center justify-center text-rose-400 shrink-0">
                <Trash2 className="w-6 h-6" />
              </div>
              <div>
                <h3 className="text-lg font-extrabold text-white">تأكيد مسح السجل من الشيت</h3>
                <p className="text-xs text-slate-400">
                  سيتم الحذف فوراً لدى جميع المستخدمين المتصلين بالشيت
                </p>
              </div>
            </div>

            <div className="bg-slate-950 border border-slate-800 rounded-2xl p-4 text-sm text-slate-200 leading-relaxed">
              هل أنتِ متأكدة من رغبتك في مسح{' '}
              <span className="font-extrabold text-rose-400">"{confirmDeleteItem.name}"</span>
              {confirmDeleteItem.ledger ? ` (دفتر رقم #${confirmDeleteItem.ledger})` : ''} نهائياً من شيت التسويات؟
            </div>

            <div className="flex items-center justify-end gap-3 pt-2">
              <button
                onClick={() => setConfirmDeleteItem(null)}
                className="px-4 py-2.5 rounded-xl text-xs font-bold bg-slate-800 hover:bg-slate-700 text-slate-200 transition cursor-pointer"
              >
                تراجع وإلغاء
              </button>
              <button
                onClick={executeDelete}
                className="px-5 py-2.5 rounded-xl text-xs font-bold bg-rose-600 hover:bg-rose-500 text-white shadow-lg shadow-rose-600/20 flex items-center gap-1.5 transition cursor-pointer"
              >
                <Trash2 className="w-4 h-4" />
                <span>نعم، امسح الآن</span>
              </button>
            </div>
          </div>
        </div>
      )}

      {/* ==========================================
          نافذة استيراد أو لصق بيانات من شيت Excel
      ========================================== */}
      {showImportModal && (
        <div className="fixed inset-0 z-50 flex items-center justify-center bg-slate-950/80 backdrop-blur-sm p-4">
          <div className="bg-slate-900 border border-slate-800 rounded-3xl max-w-2xl w-full p-6 shadow-2xl space-y-5">
            <div className="flex items-center justify-between border-b border-slate-800 pb-4">
              <div className="flex items-center gap-3">
                <div className="w-10 h-10 rounded-xl bg-teal-500/15 border border-teal-500/30 flex items-center justify-center text-teal-400">
                  <ClipboardPaste className="w-5 h-5" />
                </div>
                <div>
                  <h3 className="text-base sm:text-lg font-extrabold text-white">
                    استيراد بيانات من شيت إكسل (نسخ ولصق أو ملف CSV)
                  </h3>
                  <p className="text-xs text-slate-400">
                    انسخي الصفوف من ملف الإكسل لديكِ والصقيها هنا مباشرة ليتم إضافتها للشيت التفاعلي
                  </p>
                </div>
              </div>
              <button
                onClick={() => setShowImportModal(false)}
                className="text-slate-400 hover:text-white p-1.5 rounded-xl bg-slate-800"
              >
                <X className="w-5 h-5" />
              </button>
            </div>

            <div className="space-y-3">
              <div className="bg-slate-950 border border-slate-800 rounded-2xl p-3.5 text-xs text-slate-300 space-y-1.5">
                <p className="font-bold text-emerald-400">ترتيب الأعمدة المطلوب عند النسخ من الإكسل:</p>
                <p className="font-mono text-[11px] bg-slate-900 p-2 rounded-lg border border-slate-800 text-slate-200">
                  رقم الدفتر &nbsp;&nbsp;|&nbsp;&nbsp; اسم الموظف &nbsp;&nbsp;|&nbsp;&nbsp; ID الموظف &nbsp;&nbsp;|&nbsp;&nbsp; صافي التسوية &nbsp;&nbsp;|&nbsp;&nbsp; ملاحظات (اختياري)
                </p>
              </div>

              <div className="flex items-center justify-between">
                <label className="text-xs font-bold text-slate-300">
                  الصقي خلايا الإكسل هنا أو ارفعي ملف CSV:
                </label>
                <div>
                  <input
                    ref={fileInputRef}
                    type="file"
                    accept=".csv,.txt"
                    onChange={handleFileUpload}
                    className="hidden"
                  />
                  <button
                    type="button"
                    onClick={() => fileInputRef.current?.click()}
                    className="text-xs font-bold text-teal-400 hover:text-teal-300 bg-teal-500/10 border border-teal-500/20 px-3 py-1.5 rounded-xl flex items-center gap-1.5 cursor-pointer"
                  >
                    <Upload className="w-3.5 h-3.5" />
                    <span>اختيار ملف CSV من الجهاز</span>
                  </button>
                </div>
              </div>

              <textarea
                rows={6}
                value={pasteData}
                onChange={(e) => setPasteData(e.target.value)}
                placeholder={`مثال:\n101\tأحمد محمود علي\tEMP-901\t12500\tتمت المراجعة\n102\tمنى سعيد حسن\tEMP-902\t9800\tصرف فوري`}
                className="w-full bg-slate-950 border border-slate-700 rounded-2xl p-3.5 text-xs text-white font-mono placeholder-slate-500 outline-none focus:border-emerald-500"
              />
            </div>

            <div className="flex items-center justify-end gap-3 pt-2">
              <button
                onClick={() => setShowImportModal(false)}
                className="px-4 py-2.5 rounded-xl text-xs font-bold bg-slate-800 hover:bg-slate-700 text-slate-200 transition cursor-pointer"
              >
                إلغاء
              </button>
              <button
                disabled={isImporting || !pasteData.trim()}
                onClick={() => processImportedText(pasteData)}
                className="px-5 py-2.5 rounded-xl text-xs font-bold bg-emerald-600 hover:bg-emerald-500 disabled:opacity-50 text-white shadow-lg shadow-emerald-600/20 flex items-center gap-2 transition cursor-pointer"
              >
                {isImporting ? (
                  <RefreshCw className="w-4 h-4 animate-spin" />
                ) : (
                  <Check className="w-4 h-4" />
                )}
                <span>إدراج البيانات في الشيت الآن</span>
              </button>
            </div>
          </div>
        </div>
      )}

      {/* ==========================================
          الفوتر الرسمي الموثق بحقوق الملكية ورابط البورتفوليو (Copyright Footer)
      ========================================== */}
      <footer className="mt-12 border-t border-slate-800/80 bg-slate-900/90 backdrop-blur-md">
        <div className="max-w-7xl mx-auto px-4 sm:px-6 lg:px-8 py-6 flex flex-col sm:flex-row items-center justify-between gap-4">
          <div className="flex items-center gap-3.5">
            <div className="w-10 h-10 rounded-2xl bg-gradient-to-br from-amber-500/20 to-emerald-500/20 border border-amber-500/30 flex items-center justify-center text-amber-400 shrink-0">
              <Copyright className="w-5 h-5" />
            </div>
            <div>
              <p className="text-xs sm:text-sm font-bold text-slate-200 flex items-center flex-wrap gap-2">
                <span>جميع الحقوق محفوظة © {new Date().getFullYear()}</span>
                <span className="text-slate-600">—</span>
                <span className="text-slate-400">تصميم وتطوير:</span>
                <a
                  href={PORTFOLIO_URL}
                  target="_blank"
                  rel="noopener noreferrer"
                  title="اضغط لزيارة البورتفوليو الرسمي لـ نورهان صلاح"
                  className="group inline-flex items-center gap-1.5 text-amber-300 hover:text-white font-extrabold text-sm sm:text-base px-3 py-1 rounded-xl bg-amber-500/10 hover:bg-gradient-to-r hover:from-amber-500 hover:to-emerald-600 border border-amber-500/30 hover:border-transparent transition-all duration-300 shadow-sm hover:shadow-lg hover:shadow-amber-500/20 cursor-pointer"
                >
                  <span>{OWNER_NAME}</span>
                  <ExternalLink className="w-3.5 h-3.5 text-amber-400 group-hover:text-white transition-transform duration-300 group-hover:-translate-y-0.5 group-hover:translate-x-0.5" />
                  <span className="text-[10px] font-semibold bg-slate-950/60 group-hover:bg-black/25 text-amber-200 group-hover:text-white px-2 py-0.5 rounded-md mr-0.5">
                    زيارة البورتفوليو
                  </span>
                </a>
              </p>
            </div>
          </div>

          <div className="flex items-center gap-2.5">
            <div className="flex items-center gap-1.5 text-xs text-emerald-400 bg-emerald-950/60 border border-emerald-500/30 px-3.5 py-1.5 rounded-xl font-semibold">
              <Lock className="w-3.5 h-3.5" />
              <span>Source Code Protected | الكود المصدري محمي</span>
            </div>
          </div>
        </div>
      </footer>
    </div>
  );
}
